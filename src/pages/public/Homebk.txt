const absUrl = (u) => {
  if (!u) return null;
  if (/^https?:\/\//i.test(u)) return u;

  const base = (api?.defaults?.baseURL || "").replace(/\/api\/?$/i, "");
  return u.startsWith("/") ? `${base}${u}` : `${base}/${u}`;
};

async function safeGet(path, opts = {}) {
  try {
    const res = await api.get(path, {
      ...opts,
      validateStatus: () => true,
    });
    const status = res?.status ?? 0;
    const ok = status >= 200 && status < 300;
    return { ok, status, data: res?.data };
  } catch {
    return { ok: false, status: 0, data: null };
  }
}

function normalizeItems(data) {
  if (!data) return [];
  if (Array.isArray(data)) return data;
  if (Array.isArray(data.items)) return data.items;
  if (Array.isArray(data.videos)) return data.videos;
  if (Array.isArray(data.rows)) return data.rows;
  return [];
}

/* ---------- shared helpers for countdown ---------- */
function parseNextLiveToLocalDate(ts) {
  if (!ts) return null;
  const d = new Date(ts); // ISO/UTC → local
  if (Number.isNaN(d.getTime())) return null;
  return d;
}

function formatDuration(totalSec) {
  const s = Math.max(0, Math.floor(totalSec));
  const days = Math.floor(s / 86400);
  const hours = Math.floor((s % 86400) / 3600);
  const minutes = Math.floor((s % 3600) / 60);
  const seconds = s % 60;
  const pad = (n) => String(n).padStart(2, "0");

  return `${pad(days)}d : ${pad(hours)}h : ${pad(minutes)}m : ${pad(seconds)}s`;
}

function isObj(v) {
  return !!v && typeof v === "object" && !Array.isArray(v);
}

function uid() {
  try {
    if (typeof crypto !== "undefined" && crypto?.randomUUID)
      return crypto.randomUUID();
  } catch (_) {}
  return `id_${Math.random().toString(16).slice(2)}_${Date.now().toString(16)}`;
}

function shadowCss(v) {
  const s = String(v || "md");
  if (s === "none") return "none";
  if (s === "sm") return "0 6px 18px rgba(0,0,0,0.25)";
  if (s === "lg") return "0 18px 60px rgba(0,0,0,0.45)";
  return "0 10px 30px rgba(0,0,0,0.35)";
}

function clampNum(n, min, max) {
  const v = Number(n);
  if (Number.isNaN(v)) return min;
  return Math.max(min, Math.min(max, v));
}

function normalizeSpacing(v) {
  const s = String(v ?? "").trim();
  if (!s) return "";
  if (/^\d+(\.\d+)?$/.test(s)) return `${s}px`;
  return s;
}

function safeHtml(html) {
  // minimal guard: ensure string + fix relative /uploads URLs (React runs on :5001)
  const raw = String(html || "");
  const base = (api?.defaults?.baseURL || "").replace(/\/api\/?$/i, "");
  if (!base) return { __html: raw };

  const fixed = raw
    .replace(/(src=\")\/uploads\//g, `$1${base}/uploads/`)
    .replace(/(src=\')\/uploads\//g, `$1${base}/uploads/`)
    .replace(/(href=\")\/uploads\//g, `$1${base}/uploads/`)
    .replace(/(href=\')\/uploads\//g, `$1${base}/uploads/`);

  return { __html: fixed };
}

/* -----------------------------------------
   Public Home content loader (content_json)
----------------------------------------- */
async function fetchHomeContentJson() {
  // Try a few common shapes/endpoints without breaking anything if missing.
  const tries = [
    // ✅ primary for this backend (pagesPublic.js mounted at /api/pages)
    { path: "/pages/home" },

    // legacy / back-compat paths (older builds)
    { path: "/public/pages/home" },
    { path: "/pages/public/home" },
    { path: "/public/pages", params: { slug: "home" } },
    { path: "/public/page", params: { slug: "home" } },
  ];

  for (const t of tries) {
    const r = await safeGet(t.path, { params: t.params });
    if (!r.ok) continue;

    const d = r.data || {};
    const page = d.page || d.data?.page || d.result?.page || null;

    const cj =
      page?.content_json ||
      d?.content_json ||
      d?.page?.content_json ||
      d?.data?.content_json ||
      null;

    if (cj && isObj(cj)) return cj;

    // Sometimes endpoint returns list of pages
    const items = Array.isArray(d.pages)
      ? d.pages
      : Array.isArray(d.items)
        ? d.items
        : null;
    if (Array.isArray(items)) {
      const home = items.find(
        (x) => String(x?.slug || "").toLowerCase() === "home",
      );
      if (home?.content_json && isObj(home.content_json))
        return home.content_json;
    }
  }

  return null;
}

        if (cj && isObj(cj)) {
          setHomeModel(deepMergeHomeDefaults(cj));

        }

        if (!cancelled) setHomeLoading(false);

      cancelled = true;
    };
  }, []);

  return (
    <main className="home-root">
      <HomeSections model={homeModel} loading={homeLoading} />
      <Spacer />
    </main>
  );
}

/* ---------------------------------------------------------
   3) FEATURED HIGHLIGHT — Live Stream + Countdown
--------------------------------------------------------- */
function FeaturedHighlight() {
  const [nextLiveAt, setNextLiveAt] = useState(null);
  const [countdown, setCountdown] = useState("");
  const [showWatch, setShowWatch] = useState(false);
  const [liveOngoing, setLiveOngoing] = useState(false);

  const [loadingLive, setLoadingLive] = useState(true);

  const nextLiveAtRef = useRef(null);
  const liveOngoingRef = useRef(false);

  useEffect(() => {
    nextLiveAtRef.current = nextLiveAt;
  }, [nextLiveAt]);

  useEffect(() => {
    liveOngoingRef.current = liveOngoing;
  }, [liveOngoing]);

  const token =
    localStorage.getItem("token") || sessionStorage.getItem("token");
  const isAuthed = !!token;

  function shouldIgnoreNetworkSuspended(err) {
    const msg = String(err?.message || "");
    return msg.includes("ERR_NETWORK_IO_SUSPENDED");
  }

  function safeDate(v) {
    if (!v) return null;
    const d = new Date(v);
    if (!Number.isNaN(d.getTime())) return d;

    if (typeof v === "string") {
      const d2 = new Date(v.replace(" ", "T"));
      if (!Number.isNaN(d2.getTime())) return d2;
    }
    return null;
  }

  useEffect(() => {
    let cancelled = false;
    const controller = new AbortController();

    function setLiveState(next) {
      if (cancelled) return;
      setLiveOngoing((prev) => (prev === next ? prev : next));
      liveOngoingRef.current = next;
    }

    function setNextLiveDate(d) {
      if (cancelled) return;
      if (!d) return;

      setNextLiveAt((prev) => {
        const prevT = prev ? prev.getTime() : null;
        const nextT = d.getTime();
        if (prevT === nextT) return prev;
        return d;
      });
      nextLiveAtRef.current = d;
    }

    async function fetchNextLive() {
      if (typeof document !== "undefined" && document.hidden) return;

      try {
        const res = await api.get(`/streamcontrol/${CHANNEL_ID}/next-live`, {
          params: { _: Date.now() },
          signal: controller.signal,
        });

        const data = res.data || {};
        const state = String(data.state || "").toLowerCase();

        if (state === "live_window" || data.is_live === true) {
          setLiveState(true);
          return;
        }

        const ts =
          data.next_live_datetime ||
          data.next_live_at ||
          data.next_event?.start_at ||
          data.event?.start_at ||
          null;

        const d = safeDate(ts) || parseNextLiveToLocalDate(ts);

        if (d) {
          setLiveState(false);
          setNextLiveDate(d);
        }
      } catch (e) {
        if (controller.signal.aborted) return;
        if (shouldIgnoreNetworkSuspended(e)) return;
      }
    }

    async function fetchNextLiveFallbackAuthed() {
      if (!isAuthed) return;
      if (typeof document !== "undefined" && document.hidden) return;

      try {
        const from = new Date();
        const to = new Date();
        to.setDate(to.getDate() + 60);

        const ymd = (d) => {
          const x = new Date(d);
          const pad2 = (n) => String(n).padStart(2, "0");
          return `${x.getFullYear()}-${pad2(x.getMonth() + 1)}-${pad2(
            x.getDate(),
          )}`;
        };

        const res = await api.get("/admin/calendar/events", {
          params: { from: ymd(from), to: ymd(to), _: Date.now() },
          signal: controller.signal,
        });

        const items = Array.isArray(res.data?.events)
          ? res.data.events
          : Array.isArray(res.data)
            ? res.data
            : [];

        const now = Date.now();

        const liveEvents = items
          .filter((e) => String(e.type || "").toLowerCase() === "live")
          .map((e) => ({
            ...e,
            _start: safeDate(e.start_at),
            _end: e.end_at ? safeDate(e.end_at) : null,
          }))
          .filter((e) => e._start)
          .sort((a, b) => a._start.getTime() - b._start.getTime());

        const current = liveEvents.find(
          (e) => e._start.getTime() <= now && e._end && e._end.getTime() >= now,
        );

        if (current) {
          setLiveState(true);
          return;
        }

        const upcoming = liveEvents.find((e) => e._start.getTime() > now);
        if (upcoming) {
          setLiveState(false);
          setNextLiveDate(upcoming._start);
        }
      } catch (e) {
        if (controller.signal.aborted) return;
        if (shouldIgnoreNetworkSuspended(e)) return;
      }
    }

    async function refresh() {
      const isLive = liveOngoingRef.current;
      const hasNext = !!nextLiveAtRef.current;
      if (!isLive && !hasNext) setLoadingLive(true);

      await fetchNextLive();

      const isLive2 = liveOngoingRef.current;
      const hasNext2 = !!nextLiveAtRef.current;

      if (!cancelled && !isLive2 && !hasNext2) {
        await fetchNextLiveFallbackAuthed();
      }

      if (!cancelled) setLoadingLive(false);
    }

    refresh();
    const intv = setInterval(refresh, 15000);

    const onVis = () => {
      if (!document.hidden) refresh();
    };
    document.addEventListener("visibilitychange", onVis);

    return () => {
      cancelled = true;
      controller.abort();
      clearInterval(intv);
      document.removeEventListener("visibilitychange", onVis);
    };
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [isAuthed]);

  useEffect(() => {
    if (liveOngoing) {
      setCountdown("");
      setShowWatch(true);
      return;
    }

    if (!nextLiveAt) {
      setCountdown("");
      setShowWatch(false);
      return;
    }

    let cancelled = false;

    const tick = () => {
      if (cancelled) return;
      const now = new Date();
      const diffSec = (nextLiveAt - now) / 1000;

      if (diffSec <= 0) {
        setCountdown("");
        setShowWatch(true);
        setLiveOngoing(true);
        return;
      }

      setCountdown(formatDuration(diffSec));
      setShowWatch(false);
    };

    tick();
    const intv = setInterval(tick, 1000);

    return () => {
      cancelled = true;
      clearInterval(intv);
    };
  }, [nextLiveAt, liveOngoing]);

  return (
    <section className="feature-section">
      <img src="/media/featured-bg.jpg" alt="" className="feature-bg" />
      <div className="feature-overlay" />

      <div className="feature-content">
        <h3 className="feature-title">Live Stream</h3>

        {loadingLive && !liveOngoing && !countdown && (
          <CircularSpinner label="Loading live schedule…" />
        )}

        {countdown && !liveOngoing && (
          <div className="home-live-countdown">
            <span className="home-live-countdown-label">
              Next Live Stream Starts In
            </span>
            <span className="home-live-countdown-time">{countdown}</span>
          </div>
        )}

        {liveOngoing && (
          <div className="home-live-countdown">
            <span className="home-live-countdown-label">
              The Live Stream is ongoing
            </span>
          </div>
        )}

        {showWatch && (
          <Link to="/live-streaming" className="feature-btn">
            Watch Live
          </Link>
        )}
      </div>
    </section>
  );
}

/* ---------------------------------------------------------
   4) LATEST TEACHING — Cards only (no big banner)
   - Order: latest → oldest by publish date
   - Click goes to /watch/:id (gated handled on watch page)
--------------------------------------------------------- */
function LatestTeaching() {
  const [videos, setVideos] = useState([]);
  const [loading, setLoading] = useState(true);

  // robust publish date picker
  function getPublishTime(v) {
    const raw =
      v?.published_at ||
      v?.publish_date ||
      v?.publishedAt ||
      v?.publishedAtUtc ||
      v?.publish_at ||
      v?.created_at ||
      v?.createdAt ||
      v?.updated_at ||
      v?.updatedAt ||
      v?.metadata?.published_at ||
      v?.metadata?.publish_date ||
      v?.metadata?.publishedAt ||
      null;

    if (!raw) return 0;

    const d = new Date(raw);
    if (!Number.isNaN(d.getTime())) return d.getTime();

    // handle "YYYY-MM-DD HH:mm:ss"
    if (typeof raw === "string") {
      const d2 = new Date(raw.replace(" ", "T"));
      if (!Number.isNaN(d2.getTime())) return d2.getTime();
    }

    return 0;
  }

  useEffect(() => {
    let cancelled = false;

    (async () => {
      setLoading(true);
      try {
        const tries = [
          { path: "/public/videos", params: { limit: 12 } },
          { path: "/videos/public", params: { limit: 12 } },
        ];

        let items = [];
        for (const t of tries) {
          const r = await safeGet(t.path, { params: t.params });
          if (!r.ok) continue;
          items = normalizeItems(r.data);
          if (Array.isArray(items) && items.length) break;
        }

        const sorted = (Array.isArray(items) ? items : [])
          .slice()
          .sort((a, b) => getPublishTime(b) - getPublishTime(a));

        if (!cancelled) setVideos(sorted);
      } finally {
        if (!cancelled) setLoading(false);
      }
    })();

    return () => {
      cancelled = true;
    };
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  return (
    <section className="latest-section">
      <h2 className="latest-title">Latest Teaching</h2>
      <p className="latest-sub">A fresh impartation to strengthen your walk.</p>

      {loading ? (
        <div className="latest-loading">
          <CircularSpinner label="Loading latest videos…" />
        </div>
      ) : !videos.length ? (
        <div
          className="latest-loading"
          style={{ color: "rgba(255,255,255,0.75)" }}
        >
          No videos found yet.
        </div>
      ) : (
        <div className="latest-grid">
          {videos.map((v) => {
            const rawThumb =
              v.thumbnail_url ||
              v?.metadata?.thumbnail_vertical_url ||
              v?.metadata?.thumbnail_url ||
              null;

            const thumbSrc = absUrl(rawThumb) || DefaultThumb;
            const isDefaultThumb = !absUrl(rawThumb);

            return (
              <Link
                key={v.id}
                to={`/watch/${v.id}`}
                className="latest-card hover-card"
              >
                <div
                  className={
                    "latest-thumb" + (isDefaultThumb ? " is-default-thumb" : "")
                  }
                >
                  <img
                    src={thumbSrc}
                    alt={v.title || "Video"}
                    onError={(e) => {
                      e.currentTarget.onerror = null;
                      e.currentTarget.src = DefaultThumb;
                    }}
                  />
                  <div className="hover-play">
                    <span>▶</span>
                  </div>
                </div>

                <div className="latest-meta">
                  <h4 className="latest-video-title">{v.title}</h4>
                  <p className="latest-category">
                    {v.category_name || v?.category?.name || ""}
                  </p>
                </div>
              </Link>
            );
          })}
        </div>
      )}
    </section>
  );
}

/* ---------------------------------------------------------
   7) EXTRA SECTIONS (Builder)
   - Reflects AdminPageEditor: layout/theme/styles/html/left_html/right_html
   - Keeps your existing layout intact by using lightweight inline styling
--------------------------------------------------------- */
function ExtraSections({ sections, loading }) {
  const list = useMemo(() => {
    const arr = Array.isArray(sections) ? sections : [];
    return arr.map(ensureSectionShape).filter((s) => {
      const hasContent =
        String(s.html || "").trim() ||
        String(s.left_html || "").trim() ||
        String(s.right_html || "").trim() ||
        (Array.isArray(s.blocks) && s.blocks.length);
      // Allow title-only sections if show_title is true
      const hasTitle = !!String(s.title || "").trim() && !!s.styles?.show_title;
      return hasContent || hasTitle;
    });
  }, [sections]);

  if (!list.length) return null;

  return (
    <section style={{ paddingBottom: 10 }}>
      {loading ? null : null}

      {list.map((sec) => (
        <BuilderSection key={sec.id} sec={sec} />
      ))}
    </section>
  );
}

function renderBlocks(sec, st) {
  const blocks = Array.isArray(sec.blocks) ? sec.blocks : [];
  if (!blocks.length) return null;

  return (
    <div
      className="brtv-blocks"
      style={{
        display: "grid",
        gap: clampNum(
          String(st.content_gap || "").trim()
            ? Number(String(st.content_gap).replace(/[^0-9.]/g, "")) || 18
            : 18,
          0,
          60,
        ),
        fontSize: String(st.font_size || "").trim() || undefined,
        lineHeight: String(st.line_height || "").trim() || undefined,
      }}
    >
      {blocks.map((b) => {
        const type = String(b.type || "richtext");

        // hero / banner
        if (type === "hero") {
          const bg = absUrl(b.bg_url || "");
          const img = absUrl(b.image_url || "");
          const fit = b.bg_fit || "cover";
          const bpos = b.bg_pos || "center";
          const height = Math.max(180, Number(b.height || 420));
          const maxW = String(b.max_width || "1120px").trim() || "1120px";
          const overlay = Math.max(
            0,
            Math.min(0.85, Number(b.overlay ?? 0.35)),
          );
          const align = b.align || "left";
          const vAlign = b.v_align || "center";
          const imgPos = b.img_pos || "right";
          const fullBleed = !!b.full_bleed;

          const justifyContent =
            align === "center"
              ? "center"
              : align === "right"
                ? "flex-end"
                : "flex-start";
          const alignItems =
            vAlign === "top"
              ? "flex-start"
              : vAlign === "bottom"
                ? "flex-end"
                : "center";

          const bgImage = bg
            ? `linear-gradient(rgba(0,0,0,${overlay}), rgba(0,0,0,${overlay})), url(${bg})`
            : `linear-gradient(rgba(0,0,0,${overlay}), rgba(0,0,0,${overlay}))`;

          return (
            <section
              key={b.id}
              className={`brtv-heroBlock ${fullBleed ? "is-bleed" : ""}`}
              style={{
                minHeight: height,
                backgroundImage: bgImage,
                backgroundSize: fit,
                backgroundPosition: bpos,
              }}
            >
              <div
                className="brtv-heroInner"
                style={{ maxWidth: fullBleed ? "1400px" : maxW }}
              >
                <div
                  className="brtv-heroRow"
                  style={{
                    justifyContent,
                    alignItems,
                    flexDirection: imgPos === "left" ? "row-reverse" : "row",
                  }}
                >
                  <div
                    className="brtv-heroContent"
                    style={{ textAlign: align }}
                  >
                    {!!String(b.title || "").trim() && (
                      <h2 className="brtv-heroTitle">{b.title}</h2>
                    )}
                    {!!String(b.text || "").trim() && (
                      <p className="brtv-heroText">{b.text}</p>
                    )}
                    {!!String(b.button_text || "").trim() && (
                      <a className="brtv-heroBtn" href={b.button_href || "#"}>
                        {b.button_text}
                      </a>
                    )}
                  </div>

                  {imgPos !== "none" && img ? (
                    <div className="brtv-heroMedia">
                      <img
                        src={img}
                        alt={b.image_alt || ""}
                        className="brtv-heroImg"
                      />
                    </div>
                  ) : null}
                </div>
              </div>
            </section>
          );
        }

        if (type === "grid") {
          const cols = Math.max(1, Math.min(6, Number(b.cols) || 3));
          const gap = Math.max(0, Number(b.gap) || 12);
          const items = Array.isArray(b.items) ? b.items : [];
          return (
            <div
              key={b.id}
              className="brtv-grid"
              style={{
                display: "grid",
                gridTemplateColumns: `repeat(${cols}, minmax(0, 1fr))`,

              }}
            >

                  {String(it.image_url || "").trim() ? (

                      <img
                        alt=""

                  ) : null}

                    <div className="brtv-gridText">{it.text}</div>

              ))}

        if (type === "gallery") {

function BuilderSection({ sec }) {
  const st = sec.styles || defaultSectionStyle(sec.theme);
  const t = themeDefaults(sec.theme);

  // Layout mapping (kept minimal so your CSS doesn’t get disrupted)
  const isFull = sec.layout === "full";
  const isHighlight = sec.layout === "highlight";
  const isTwo = sec.layout === "two_columns";
  const hasBlocks = Array.isArray(sec.blocks) && sec.blocks.length > 0;

  const maxW = String(st.max_width || "").trim();
  const wrapMaxWidth = maxW || (isFull ? "100%" : "1100px");

  const pad = st.padding || (isFull ? "48px 0" : "22px 18px");
  const mg = st.margin || (isFull ? "0" : "18px auto");

  const radius = clampNum(st.radius ?? 14, 0, 60);

  const bgColor =
    String(st.bg_color || "").trim() ||
    (isHighlight ? "rgba(154, 92, 255, 0.12)" : t.bg);

  const borderColor = String(st.border_color || "").trim() || t.border;

  const textColor = String(st.text_color || "").trim() || t.text;

  const titleColor = String(st.title_color || "").trim() || textColor;

  const overlay = Math.max(0, Math.min(0.9, Number(st.overlay ?? 0) || 0));

  const bgImg =
    absUrl(st.bg_image_url) || (st.bg_image_url ? st.bg_image_url : null);

  const bgPosition = String(st.bg_image_pos || "center");
  const bgFit = String(st.bg_image_fit || "cover");

  const borderW = String(st.border_width || "").trim();
  const overlayOpacityNumRaw =
    String(st.overlay_opacity ?? "").trim() === ""
      ? null
      : Number(st.overlay_opacity);
  const overlayOpacityNum = Number.isFinite(overlayOpacityNumRaw)
    ? Math.max(0, Math.min(0.9, overlayOpacityNumRaw))
    : overlay;

  const overlayColor = String(st.overlay_color || "").trim();
  const overlayLayer = overlayColor
    ? `linear-gradient(${overlayColor}, ${overlayColor})`
    : `linear-gradient(rgba(0,0,0,${overlayOpacityNum}), rgba(0,0,0,${overlayOpacityNum}))`;

  const contentGap = String(st.content_gap || "").trim()
    ? String(st.content_gap).trim()
    : "12px";

  const outerStyle = {
    margin: mg,
    maxWidth: wrapMaxWidth,
  };

  const innerStyle = {
    position: "relative",
    padding: pad,
    minHeight: String(st.min_height || "").trim() || undefined,
    height: String(st.height || "").trim() || undefined,
    borderRadius: radius,
    color: textColor,
    background: bgImg ? `${overlayLayer}, url(${bgImg})` : bgColor,
    backgroundSize: bgImg ? bgFit : undefined,
    backgroundPosition: bgImg ? bgPosition : undefined,
    backgroundRepeat: bgImg ? "no-repeat" : undefined,
    border: `${borderW || "1px"} solid ${borderColor}`,
    boxShadow: shadowCss(st.shadow),
    textAlign: String(st.align || "left"),
    overflow: "hidden",
    // don’t force width changes; let Home.css do the layout
  };

  const contentMax = isFull ? "1100px" : "100%";

  const Tag = String(st.title_tag || "h3");

  const titleStyle = {
    color: titleColor,
    fontSize: clampNum(st.title_size ?? 22, 10, 64),
    fontWeight: clampNum(st.title_weight ?? 800, 200, 900),
    margin: 0,
    marginBottom: contentGap,
    letterSpacing: -0.2,
  };

  const imgRadius = clampNum(st.img_radius ?? 12, 0, 60);
  const imgShadow =
    st.img_shadow === "soft" ? "0 14px 50px rgba(0,0,0,0.45)" : "none";

  const scopedCss = `
    #sec-${sec.id} img {
      border-radius: ${imgRadius}px;
      box-shadow: ${imgShadow};
    }
    #sec-${sec.id} a { color: inherit; }
  `;

  return (
    <div style={outerStyle}>
      <div id={`sec-${sec.id}`} style={innerStyle}>
        <style>{scopedCss}</style>

        <div
          style={{
            maxWidth: contentMax,
            margin: "0 auto",
            fontSize: String(st.font_size || "").trim() || undefined,
            lineHeight: String(st.line_height || "").trim() || undefined,
          }}
        >
          {st.show_title !== false && String(sec.title || "").trim() ? (
            <Tag style={titleStyle}>{sec.title}</Tag>
          ) : null}

          {hasBlocks ? (
            renderBlocks(sec, st)
          ) : isTwo ? (
            <div
              style={{
                display: "grid",
                gridTemplateColumns: "1fr 1fr",
                gap: 18,
                alignItems: "start",
              }}
            >
              <div
                style={{ minWidth: 0 }}
                dangerouslySetInnerHTML={safeHtml(sec.left_html || "")}
              />
              <div
                style={{ minWidth: 0 }}
                dangerouslySetInnerHTML={safeHtml(sec.right_html || "")}
              />
            </div>
          ) : (
            <div dangerouslySetInnerHTML={safeHtml(sec.html || "")} />
          )}
        </div>
      </div>
    </div>
  );
}
