// src/layout/PublicLayout.jsx
import React, { useEffect, useMemo, useState } from "react";
import { NavLink, Outlet, useLocation, useNavigate } from "react-router-dom";
import "./PublicLayout.css";
import useSupportWidget from "../support/useSupportWidget";
import api from "../api";
import SiteFooter from "../components/SiteFooter";

export default function PublicLayout() {
  useSupportWidget();

  const [open, setOpen] = useState(false); // mobile menu
  const [me, setMe] = useState(null);
  const location = useLocation();
  const navigate = useNavigate();

  // Hide Bishop TV global header on channel routes (/c, /c/:slug, /c/:slug/*)
  const isChannelRoute = /^\/c(\/|$)/.test(location.pathname);

  const token = useMemo(
    () => localStorage.getItem("token") || sessionStorage.getItem("token"),
    // eslint-disable-next-line react-hooks/exhaustive-deps
    [location.key]
  );

  // Close menu on route change
  useEffect(() => {
    setOpen(false);
  }, [location.pathname]);

  // Lock body scroll on mobile menu
  useEffect(() => {
    if (typeof document === "undefined") return;
    const prev = document.body.style.overflow;
    document.body.style.overflow = open ? "hidden" : prev || "";
    return () => {
      document.body.style.overflow = prev || "";
    };
  }, [open]);

  // ESC to close
  useEffect(() => {
    if (!open) return;
    const onKey = (e) => {
      if (e.key === "Escape") setOpen(false);
    };
    window.addEventListener("keydown", onKey);
    return () => window.removeEventListener("keydown", onKey);
  }, [open]);

  // Load current user (to show Admin/Account links)
  useEffect(() => {
    let cancelled = false;
    async function loadMe() {
      if (!token) {
        setMe(null);
        return;
      }
      try {
        const { data } = await api.get("/auth/me");
        if (!cancelled) setMe(data?.user || data || null);
      } catch {
        if (!cancelled) setMe(null);
      }
    }
    loadMe();
    return () => {
      cancelled = true;
    };
  }, [token]);

  const isAdmin = (me?.role || me?.type || "").toLowerCase() === "admin";
  const isAuthed = !!token;

  function logout() {
    localStorage.removeItem("token");
    sessionStorage.removeItem("token");
    setMe(null);
    navigate("/login", { replace: true });
  }

  const linkClass = ({ isActive }) => "pl-link" + (isActive ? " active" : "");

  return (
    <div className={`pl-shell ${open ? "nav-open" : ""}`}>
      {/* Bishop TV global header — hidden on channel pages */}
      {!isChannelRoute && (
        <>
          <header className="pl-header">
            <div className="pl-brand">
              <NavLink to="/" className="pl-logo">
                <span className="pl-logo-dot">B</span>
                <span className="pl-logo-text">
                  Bishop<span className="accent">TV</span>
                </span>
              </NavLink>
              <button
                className="pl-burger"
                aria-label="Open menu"
                aria-expanded={open}
                onClick={() => setOpen((v) => !v)}
              >
                <svg width="22" height="22" viewBox="0 0 24 24" fill="none">
                  <path
                    d="M4 6h16M4 12h16M4 18h16"
                    stroke="currentColor"
                    strokeWidth="2"
                    strokeLinecap="round"
                  />
                </svg>
              </button>
            </div>

            <nav className="pl-nav">
              {/* primary nav left (kept empty for now) */}
            </nav>

            <div className="pl-actions">
              <nav className="pl-nav">
                <NavLink end to="/" className={linkClass}>
                  Home
                </NavLink>
                <NavLink to="/catalog" className={linkClass}>
                  Watch
                </NavLink>
                {/* ✅ NEW: Community link */}
                <NavLink to="/community" className={linkClass}>
                  Community
                </NavLink>
                <NavLink to="/become-a-partner" className={linkClass}>
                  Become a Partner
                </NavLink>
                <NavLink to="/store" className={linkClass}>
                  Store
                </NavLink>
                <NavLink to="/give" className={linkClass}>
                  Give
                </NavLink>
              </nav>

              {isAdmin && (
                <NavLink to="/admin" className="pl-btn ghost">
                  Admin
                </NavLink>
              )}
              {isAuthed ? (
                <>
                  <NavLink to="/account" className="pl-btn outline">
                    My Account
                  </NavLink>
                  <button className="pl-btn ghost" onClick={logout}>
                    Logout
                  </button>
                </>
              ) : (
                <>
                  <NavLink to="/login" className="pl-btn ghost">
                    Login
                  </NavLink>
                  {/* Sign Up → Start for Free */}
                  <NavLink to="/free-account" className="pl-btn outline">
                    Start for Free
                  </NavLink>
                </>
              )}
            </div>
          </header>

          {/* Mobile drawer */}
          <div className="pl-drawer">
            <NavLink
              end
              to="/"
              className="pl-drawer-link"
              onClick={() => setOpen(false)}
            >
              Home
            </NavLink>
            {/* ✅ NEW: Community in mobile drawer */}
            <NavLink
              to="/community"
              className="pl-drawer-link"
              onClick={() => setOpen(false)}
            >
              Community
            </NavLink>
            <NavLink
              to="/catalog"
              className="pl-drawer-link"
              onClick={() => setOpen(false)}
            >
              Watch
            </NavLink>
            <NavLink
              to="/p/events"
              className="pl-drawer-link"
              onClick={() => setOpen(false)}
            >
              Events
            </NavLink>
            <NavLink
              to="/pricing"
              className="pl-drawer-link"
              onClick={() => setOpen(false)}
            >
              Pricing
            </NavLink>

            <div className="pl-drawer-sep" />

            {isAdmin && (
              <NavLink
                to="/admin"
                className="pl-drawer-link"
                onClick={() => setOpen(false)}
              >
                Admin
              </NavLink>
            )}

            {isAuthed ? (
              <>
                <NavLink
                  to="/account"
                  className="pl-drawer-link"
                  onClick={() => setOpen(false)}
                >
                  My Account
                </NavLink>
                <button
                  className="pl-drawer-link btnlike"
                  onClick={() => {
                    setOpen(false);
                    logout();
                  }}
                >
                  Logout
                </button>
              </>
            ) : (
              <>
                <NavLink
                  to="/login"
                  className="pl-drawer-link"
                  onClick={() => setOpen(false)}
                >
                  Login
                </NavLink>
                {/* Sign Up → Start for Free */}
                <NavLink
                  to="/free-account"
                  className="pl-drawer-link"
                  onClick={() => setOpen(false)}
                >
                  Start for Free
                </NavLink>
              </>
            )}
          </div>

          {open && (
            <button
              className="pl-dim"
              onClick={() => setOpen(false)}
              aria-label="Close menu"
            />
          )}
        </>
      )}

      {/* Full-width content */}
      <main className="pl-main">
        <Outlet />
      </main>
      <SiteFooter />
    </div>
  );
}
